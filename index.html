Pengaturan Webhook Google Sheets

Hubungkan aplikasi dengan database Google Sheets melalui Google Apps Script Web App

Kode Google Apps Script (Code.gs) Copy paste kode ini di Extensions > Apps Script spreadsheet Anda 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    data.timestamp || new Date(),
    data.nip,
    data.nama,
    data.tipe,
    data.status,
    data.lokasi,
    data.keterangan
  ]);
  return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
